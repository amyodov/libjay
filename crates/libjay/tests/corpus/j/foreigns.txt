// J's foreigns, `m !: n` — the members that only COMPUTE.
//
// A foreign that reaches a file, a script, the host, the clock or a shared
// library is closed by the sandbox and is refused where it is written, so
// none of those is here. What is here is the arithmetic ones: the `3!:`
// byte conversions, the `4!:` name table, the `5!:` representations, the
// `8!:` formats, the two `9!:` parameters libjay honours, and `128!:3`.
//
// A program is one line with `\n` for the newlines in it.

// --- 5!:1 and 5!:0, the atomic representation and its inverse -------------
f =: +/ % #\n5!:1 <'f'
n =: 5\n5!:1 <'n'
f =: +/ % #\nh =: (5!:1 <'f') 5!:0\nh 1 2 3 4
f =: *:@:+\nh =: (5!:1 <'f') 5!:0\nh 3
n =: 7\nk =: (5!:1 <'n') 5!:0\nk
f =: 3 &+\nh =: (5!:1 <'f') 5!:0\nh 4

// --- 5!:2, the boxed representation ---------------------------------------
n =: 5\n5!:2 <'n'
n =: 1 2 3\n5!:2 <'n'
f =: +/\n5!:2 <'f'
f =: +/ % #\n5!:2 <'f'
f =: +/ @: *:\n5!:2 <'f'
f =: 3 &+\n5!:2 <'f'
f =: 2 + 3 * ]\n5!:2 <'f'
f =: <@:(+/)\n5!:2 <'f'
5!:2 <'nosuch'

// --- 5!:5, the linear representation --------------------------------------
n =: 5\n5!:5 <'n'

f =: +/\\\n5!:5 <'f'

f =: 3 : 'y+1'\n5!:5 <'f'
f =: 4 : 'x + y'\n5!:5 <'f'

// --- 5!:6, the parenthesised representation -------------------------------

f =: (+/ % #) @: *:\n5!:6 <'f'

f =: 2 + 3 * ]\n5!:6 <'f'
f =: <@:(+/)\n5!:6 <'f'
f =: 3 : 'y+1'\n5!:6 <'f'
