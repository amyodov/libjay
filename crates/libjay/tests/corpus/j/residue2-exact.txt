// Where the EXACT types survive a verb and where they do not.
//
// The matrix divide takes its exactness from the SYSTEM — the right
// argument, which is the one inverted — and the one-unknown system an atom
// names follows the same rule. The set difference keeps the LEFT argument's
// type against an ATOM and widens against a list. The conversion back from
// an exact type and the prime queries both leave the boolean empty where
// the argument had nothing in it.

// `x %. y`: the inverse is rational only where y is extended or rational,
// and the product with x promotes from there.
(123x) %. 2
(_5x) %. 2
(1r2 1r3) %. 2
3!:0 ((123x) %. 2)
3!:0 ((1r2 1r3) %. 2)
2 %. (123x)
3!:0 (2 %. (123x))
(123x) %. (2x)
3!:0 ((123x) %. (2x))
0.5 %. (2x)
3!:0 (0.5 %. (2x))
(2j1) %. (2x)
3!:0 ((2j1) %. (2x))
3!:0 ((1 2 3x) %. (1 2 3))
3!:0 ((2x) %. (2j1))
3!:0 ((2x) %. (0.5))
3!:0 (%. (123x))
3!:0 (%. (2 2 $ 1x 2x 3x 4x))

// `x -. y`: an ATOM on the right leaves the left argument's type alone; a
// LIST of one item widens it by the ordinary promotion.
(1r2 1r3) -. (1ar1)
3!:0 ((1r2 1r3) -. (1ar1))
3!:0 ((1r2 1r3) -. (0j1))
3!:0 ((1r2 1r3) -. (2.5))
3!:0 ((1r2 1r3) -. (2))
3!:0 ((1x 2x) -. (2.5))
3!:0 ((1x 2x) -. (0j1))
3!:0 ((1 2) -. (2.5))
3!:0 ((1 2) -. (1r2))
3!:0 ((2.5 3.5) -. (1r2))
3!:0 ((0j1 1) -. (1r2))
3!:0 ((1r2 1r3) -. (''))
3!:0 ((1r2 1r3) -. (_ __ 0))
3!:0 ((1r2 1r3) -. (_))
3!:0 ((1x 2x) -. (_))
3!:0 ((1r2 1r3) -. (_.))
3!:0 ((1r2 1r3) -. (,2.5))
3!:0 ((1r2 1r3) -. (2.5 3.5))
3!:0 ((1r2 1r3) -. (1r2 1r4))
3!:0 ((1r2 1r3) -. (2 3))
3!:0 ((1r2 1r3) -. (,0j1))
3!:0 ((1x 2x) -. (,2.5))
(1x 2x 3x) -. (2x)
3!:0 ((1x 2x 3x) -. (2x))

// `_1 x:` over an argument with nothing in it: the boolean empty, whatever
// the empty was written as.
3!:0 (_1 x: (0 $ 'a'))
3!:0 (_1 x: (0 $ 0))
3!:0 (_1 x: (0 $ <0))
$ (_1 x: (0 $ <0))
3!:0 (_1 x: (0 $ 0j1))
3!:0 (_1 x: (0 $ 1x))
_1 x: (1r2 1r3)
_1 x: (2 3x)
3!:0 (_1 x: (2 3x))

// A prime query over an argument with no items leaves the boolean empty
// too, where one number with no factor keeps the type it was written in.
3 p: 0
3 p: 1
3!:0 (3 p: 1)
3!:0 (3 p: (2 0 3 $ 0))
3!:0 (3 p: (0 $ 0))
$ (3 p: (2 0 3 $ 0))
3!:0 (3 p: (2 0 3 $ 1))
3!:0 (3 p: (0 $ 1x))
3!:0 (3 p: (0 $ 0.5))
3!:0 (3 p: (0 $ (a:)))
3!:0 (q: (0 $ 0))
3!:0 (q: (0 $ 1x))
3!:0 (q: 1x)
3!:0 (q: 1)
