// Explicit adverbs and conjunctions: `1 : '…'`, `2 : '…'`, `1 : 0` and
// `2 : 0`, and the `{{ … }}` whose body names an operand.
//
// An adverb takes the verb written to its LEFT; a conjunction sits between
// its two operands. The operands arrive as `u` and `v` when they are verbs
// and as `m` and `n` when they are nouns.

// --- 1 : '…', a body that names y: it becomes the derived verb's body
twice =. 1 : 'u u y'\n+: twice 3
twice =. 1 : 'u u y'\n>: twice 5
twice =. 1 : 'u u y'\n*: twice 3
twice =. 1 : 'u u y'\n+: twice i. 4
+: (1 : 'u u y') 3
sum =. 1 : 'u/ y'\n+ sum 1 2 3 4
under =. 1 : 'u y'\n- under 7
selfd =. 1 : 'y u y'\n+ selfd 6

// --- a body that names x: the derived verb is a dyad and nothing else
addy =. 1 : 'x u y'\n3 (+ addy) 4
addy =. 1 : 'x u y'\n2 (* addy) 5
twice =. 1 : 'u u y'\n3 (+: twice) 4
addy =. 1 : 'x u y'\n(+ addy) 4

// --- 2 : '…', two operands
pair =. 2 : 'u v y'\n(+: pair *:) 3
pair =. 2 : 'u v y'\n(- pair >:) 4
both =. 2 : 'x u v y'\n3 (+ both *:) 4
both =. 2 : 'x u v y'\n10 (- both -:) 4

// --- noun operands answer to m and n
addm =. 1 : 'm + y'\n5 addm 3
addm =. 1 : 'm + y'\n2 addm i. 4
joinn =. 2 : 'm , y , n'\n(2 joinn 3) 9
// the name has to match the operand's part of speech
f =. 1 : 'u y'\n5 f 3
f =. 1 : 'm + y'\n+: f 3

// --- a body that names NEITHER x nor y runs at derivation, and what it
// makes is what the modifier produced: usually a tacit verb, sometimes a
// noun.
sq =. 1 : 'u @ u'\n+: sq 3
sq =. 1 : 'u @ u'\n+: sq i. 4
sq =. 1 : 'u @ u'\nd =. +: sq\nd 5
half =. 1 : 'u & 2'\n% half 8
comp =. 2 : 'u @: v'\n(+: comp *:) 3
lit =. 1 : '3 + 4'\n+: lit

// --- {{ … }}: the words of the body decide the part of speech
tw =. {{ u u y }}\n+: tw 3
tw =. {{ u u y }}\n>: tw 5
cj =. {{ u v y }}\n(+: cj *:) 3
cj =. {{ x u v y }}\n3 (+ cj *:) 4
nm =. {{ m + y }}\n5 nm 3
nn =. {{ m , y , n }}\n(1 nn 9) 5
+: ({{ u u y }}) 3

// --- {{)a and its relatives state the part of speech outright
mk =. {{)a\nu u y\n}}\n+: mk 3
mk =. {{)c\nu v y\n}}\n(+: mk *:) 3
mk =. {{)v\ny + 1\n}}\nmk 4
mk =. {{)d\nx + y\n}}\n3 mk 4
mk =. {{)m\n- y\n}}\nmk 4

// --- the body on the lines below, ending with `)`
tw =. 1 : 0\nu u y\n)\n+: tw 3
cj =. 2 : 0\nx u v y\n)\n3 (+ cj *:) 4
ml =. 1 : 0\nz =. u y\nz + z\n)\n+: ml 3
ml =. 2 : 0\nif. x > 0 do. u y else. v y end.\n)\n1 (+: ml *:) 5
ml =. 2 : 0\nif. x > 0 do. u y else. v y end.\n)\n0 (+: ml *:) 5

// --- a derived verb composes with everything a verb composes with
tw =. 1 : 'u u y'\n(+: tw)"0 i. 3
tw =. 1 : 'u u y'\n(+:"0 tw) i. 3
tw =. 1 : 'u u y'\n(>: @ (+: tw)) 3
tw =. 1 : 'u u y'\nh =. +: tw\n(h , h) 3
ins =. 1 : 'u/ y'\nmean =. (+ ins) % #\nmean 1 2 3 4

// --- nesting: a modifier in another modifier's body, and a derived verb
// as another derivation's operand
tw =. 1 : 'u u y'\nq =. 1 : '(u tw) tw'\n+: q 1
tw =. 1 : 'u u y'\ng =. 2 : 'u tw v y'\n(+: g *:) 3
tw =. 1 : 'u u y'\n+: tw +: tw 1
tw =. 1 : 'u u y'\n(+: tw) tw 1

// --- the body reads the names the program has already given
q =. 10\nf =. 1 : 'q + u y'\n+: f 3
// and the operand reaches every place a verb goes in it
f =. 1 : 'u^:2 y'\n>: f 3
f =. {{ u&2 y }}\n^ f 3
f =. 2 : 'x u v y'\n1 2 3 (+ f *:) 4
twice =. 1 : 'u u y'\nthrice =. 1 : 'u u u y'\n+: thrice twice 1
// a modifier name can be given a second name
f =. 1 : 'u y'\ng =. f\n+: g 3

// --- a power count the program computes ----------------------------------
// `u^:n` reads n where the derived verb is APPLIED, so a name, an
// expression, or a definition's own argument may decide it.
n =. 3\n(>:^:n) 0
n =. 3\n>:^:n ] 0
n =. 0\n>:^:n ] 0
n =. 2 3\n>:^:n ] 0
n =. 2\n+:^:n ] 3
n =. 4\n+:^:n ] 3
(>:^:(1+2)) 0
(+:^:(1+1)) 3
n =. 1\n(+:^:(n+2)) 3
n =. _1\n(>:^:n) 5
f =. 3 : '>:^:y ] 0'\nf 4
f =. 3 : '>:^:y ] 0'\nf 0
f =. 3 : '+:^:y ] 1'\nf 5
f =. 4 : '+:^:x ] y'\n3 f 1
n =. <3\n>:^:n ] 0
