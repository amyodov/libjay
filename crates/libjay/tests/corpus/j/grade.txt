// Grading a BOXED array: J's total array ordering, one line per rule.
// Recorded from jconsole; every claim below is the oracle's, not a guess.

// The type class first: numeric, then character, then boxed.
/: 1;'a';(1 2);(<<3)
\\: 1;'a';(1 2);(<<3)
/: 'b';1;'a';2
/: (<1);(<'a')
/: 1;(<1)
/: 'a';1
/: (<a:),(<1)
/: (<a:),(<'')
/: (<'ab'),(<1 2 3)
/: (<1j1),(<'a')

// Then the rank, ascending — a class outranks it, a shape does not.
/: (i.0);1
/: (1 1$1);(1 2)
/: (2 2$0);(1 2 3 4)
/: (2 2 2$0);(3 3$0)
/: (<i.0),(<2 2$1)
/: (<'a'),(<'ab')

// Then the shape, read with the LAST axis most significant.
/: (2 3$0);(3 2$0)
/: (1 6$0);(6 1$0)
/: (0 3$0);(3 0$0)
/: (1 4$0);(2 3$0)
/: (1 4$0);(2 2$0)
/: (2 1 3$0);(1 2 3$0)
/: (3 1$0);(1 3$0)
/: (<2 0$0),(<0 2$0)
/: (1 2);(1 2 3)
/: 'abc';'ab'

// Then the atoms, in row-major order. A boxed atom recurses.
/: 'abc';'abd'
/: (<1 2 3),(<1 2 4),(<1 2)
/: (<2 2$0 1 2 3),(<2 2$0 1 2 4)
/: (<<1 2),(<<1 3)
/: (<<1 2),(<<1)
/: (<<'a'),(<<1)
/: (<<1),(<<<1)
/: (<1;2),(<'a';'b')
/: (<'a'),(<'a'),(<'A')

// The atoms compare exactly, across the numeric tower, and a NaN ties
// with everything.
/: (<1),(<1+1e_14)
/: (<1+1e_14),(<1)
/: (<1r3),(<0.3333333333)
/: 2,(1+1e_14),1
/: (<1x),(<2)
/: (<1j2),(<1j1)
/: (<__),(<0),(<_)
/: (<_.),(<1)

// An EMPTY array has no atoms to take a class from, so it takes the
// lowest one whatever its type; its rank and shape still count.
/: (<0$'a'),(<i.0)
/: (<i.0),(<0$'a')
/: (<''),(<'a')
/: (<''),(<1)
/: (<''),(<<1)
/: (<i.0),(<<1)
/: (<0$0),(<0$a:),(<0$'a')
/: (<'a'),(<1 2),(<'')
/: (<0$<0),(<1 2 3)

// Equal items keep their order, up and down alike.
/: (<'a'),(<'a'),(<'b')
\\: (<'a'),(<'a'),(<'b')
\\: (<1),(<1),(<2)

// A boxed array of rank 2 grades its ROWS.
/: 2 2$(<1),(<2),(<0),(<3)

/: 3 1$(<1),(<'a'),(<<1)
